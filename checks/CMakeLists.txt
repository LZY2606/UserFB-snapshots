cmake_minimum_required(VERSION 3.21)
project(check_support LANGUAGES C)
set(CMAKE_C_STANDARD 23)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(XM_VERBOSE OFF CACHE BOOL "" FORCE)
add_subdirectory(../src xm_build)
add_executable(check_baseline baseline.c)
target_link_libraries(check_baseline PRIVATE xm m)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../tests/stereo_samples.c")
  add_executable(check_target driver.c ../tests/stereo_samples.c)
  target_include_directories(check_target PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
  target_link_libraries(check_target PRIVATE xm m)
endif()
