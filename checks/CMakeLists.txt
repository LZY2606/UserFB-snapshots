cmake_minimum_required(VERSION 3.15)
project(filter_contract_support LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LUNASVG_BUILD_EXAMPLES ON CACHE BOOL "" FORCE)
set(LUNASVG_DISABLE_LOAD_SYSTEM_FONTS ON CACHE BOOL "" FORCE)
set(LUNASVG_DISABLE_EXTERNAL_RESOURCES ON CACHE BOOL "" FORCE)
add_subdirectory(.. upstream)
add_executable(filter_baseline current.cpp)
target_link_libraries(filter_baseline PRIVATE lunasvg::lunasvg)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../tests/filter_contract.cpp")
    add_executable(filter_contract ../tests/filter_contract.cpp)
    target_link_libraries(filter_contract PRIVATE lunasvg::lunasvg)
endif()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../examples/filter_demo.cpp")
    add_executable(filter_demo ../examples/filter_demo.cpp)
    target_link_libraries(filter_demo PRIVATE lunasvg::lunasvg)
endif()
