cmake_minimum_required(VERSION 3.16.3)

set(PMP_BUILD_TESTS ON CACHE BOOL "" FORCE)
set(PMP_INSTALL OFF CACHE BOOL "" FORCE)
set(PMP_BUILD_DOCS OFF CACHE BOOL "" FORCE)
set(PMP_STRICT_COMPILATION OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(EIGEN3_INCLUDE_DIR "${CMAKE_CURRENT_LIST_DIR}/../external/eigen-5.0.1" CACHE PATH "" FORCE)
add_subdirectory(.. pmp-build)
add_executable(check_baseline baseline.cpp)
target_link_libraries(check_baseline PRIVATE pmp)
