cmake_minimum_required(VERSION 3.16)
project(check_mlt C)
include(GNUInstallDirs)
find_package(Threads REQUIRED)
set(MLT_VERSION 7.6.0)
set(MLT_VERSION_MAJOR 7)
set(MLT_INSTALL_DATA_DIR "${CMAKE_BINARY_DIR}/data")
set(MLT_COMPILE_OPTIONS "")
add_subdirectory(../src/framework framework)
add_library(check_mix STATIC ../src/modules/core/transition_mix.c)
target_link_libraries(check_mix PUBLIC mlt m)
add_executable(baseline baseline.c)
target_link_libraries(baseline PRIVATE check_mix)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../src/tests/sample_curve.c")
 add_executable(target_tests test_main.c ../src/tests/sample_curve.c)
 target_include_directories(target_tests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
 target_link_libraries(target_tests PRIVATE check_mix)
endif()
