cmake_minimum_required(VERSION 3.16)
project(TaskImaSupport LANGUAGES C CXX)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
foreach(option BUILD_TESTING BUILD_PROGRAMS BUILD_EXAMPLES BUILD_REGTEST ENABLE_EXTERNAL_LIBS ENABLE_MPEG ENABLE_CPACK ENABLE_PACKAGE_CONFIG INSTALL_PKGCONFIG_MODULE)
 set(${option} OFF CACHE BOOL "" FORCE)
endforeach()
set(CMAKE_DISABLE_FIND_PACKAGE_Python TRUE CACHE BOOL "" FORCE)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/.." upstream)
if(TASK_TARGET)
 if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../src/wavl.c")
  message(FATAL_ERROR "Required production source src/wavl.c is missing")
 endif()
 target_sources(sndfile PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../src/wavl.c")
 target_compile_options(sndfile PRIVATE -include "${CMAKE_CURRENT_SOURCE_DIR}/../src/wavl.h")
endif()
add_executable(baseline baseline.c)
target_include_directories(baseline PRIVATE .. ../include)
target_link_libraries(baseline PRIVATE sndfile)
