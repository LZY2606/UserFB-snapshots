cmake_minimum_required(VERSION 3.5)
project(metadata_support LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(BW64_EXAMPLES ON CACHE BOOL "" FORCE)
set(BW64_UNIT_TESTS ON CACHE BOOL "" FORCE)
set(BW64_PACKAGE_AND_INSTALL OFF CACHE BOOL "" FORCE)
get_filename_component(task_root "${CMAKE_CURRENT_LIST_DIR}/.." ABSOLUTE)
add_subdirectory("${task_root}" upstream)
add_executable(production_witness witness.cpp)
target_link_libraries(production_witness PRIVATE bw64)
if(EXISTS "${task_root}/tests/metadata_edit_target.cpp")
  set_source_files_properties(driver.c PROPERTIES LANGUAGE CXX)
  add_executable(task_target driver.c "${task_root}/tests/metadata_edit_target.cpp")
  target_include_directories(task_target PRIVATE "${task_root}")
  target_link_libraries(task_target PRIVATE bw64)
endif()
if(EXISTS "${task_root}/examples/metadata_edit.cpp")
  add_executable(metadata_edit_example "${task_root}/examples/metadata_edit.cpp")
  target_link_libraries(metadata_edit_example PRIVATE bw64)
endif()
