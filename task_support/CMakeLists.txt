cmake_minimum_required(VERSION 3.12...3.31)
project(check_support C CXX)
set(WITH_OGG OFF CACHE BOOL "" FORCE)
set(ENABLE_MULTITHREADING OFF CACHE BOOL "" FORCE)
set(BUILD_CXXLIBS OFF CACHE BOOL "" FORCE)
set(BUILD_PROGRAMS OFF CACHE BOOL "" FORCE)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_DOCS OFF CACHE BOOL "" FORCE)
set(INSTALL_MANPAGES OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING ON CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(CMAKE_DISABLE_FIND_PACKAGE_Intl TRUE)
set(CMAKE_DISABLE_FIND_PACKAGE_Iconv TRUE)
get_filename_component(UPSTREAM "${CMAKE_CURRENT_LIST_DIR}/.." ABSOLUTE)
add_subdirectory("${UPSTREAM}" flac)
add_executable(check_baseline baseline.c)
target_link_libraries(check_baseline PRIVATE FLAC)
add_executable(check_old_encoders old_encoders.c "${UPSTREAM}/src/test_libFLAC/encoders.c")
target_compile_definitions(check_old_encoders PRIVATE HAVE_CONFIG_H)
target_include_directories(check_old_encoders PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/flac" "${UPSTREAM}/include" "${UPSTREAM}/src")
target_link_libraries(check_old_encoders PRIVATE FLAC grabbag test_libs_common)
if(CHECK_TARGET)
  if(NOT EXISTS "${UPSTREAM}/tests/adaptive_blocks.c")
    message(FATAL_ERROR "Create tests/adaptive_blocks.c first")
  endif()
  add_executable(check_target driver.c "${UPSTREAM}/tests/adaptive_blocks.c")
  target_compile_definitions(FLAC PRIVATE FLAC__TEST_BLOCKSIZE_SEARCH)
  target_compile_definitions(check_target PRIVATE FLAC__TEST_BLOCKSIZE_SEARCH)
  target_include_directories(check_target PRIVATE "${CMAKE_CURRENT_LIST_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/flac" "${UPSTREAM}/src/libFLAC/include")
  target_link_libraries(check_target PRIVATE FLAC)
endif()
