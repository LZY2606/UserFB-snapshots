cmake_minimum_required(VERSION 3.16)
project(TaskNsfSupport LANGUAGES C CXX)
set(GME_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(GME_BUILD_STATIC ON CACHE BOOL "" FORCE)
set(GME_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GME_BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(GME_ZLIB OFF CACHE BOOL "" FORCE)
foreach(codec AY GBS GYM HES KSS NSFE SAP SPC VGM)
  set(USE_GME_${codec} OFF CACHE BOOL "" FORCE)
endforeach()
set(USE_GME_NSF ON CACHE BOOL "" FORCE)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/.." upstream)
if(TASK_TARGET)
  target_sources(gme_static PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../gme/Nsf_Pump.cpp")
  target_compile_options(gme_static PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-include> $<$<COMPILE_LANGUAGE:CXX>:${CMAKE_CURRENT_SOURCE_DIR}/../gme/Nsf_Pump.h>)
  add_executable(api_consumer api_consumer.cpp)
  target_compile_features(api_consumer PRIVATE cxx_std_17)
  target_include_directories(api_consumer PRIVATE .. ../gme)
  target_compile_options(api_consumer PRIVATE -fno-rtti)
  target_link_libraries(api_consumer PRIVATE gme_static)
endif()
add_executable(baseline baseline.cpp)
target_include_directories(baseline PRIVATE .. ../gme)
target_compile_options(baseline PRIVATE -fno-rtti)
target_compile_features(baseline PRIVATE cxx_std_17)
target_link_libraries(baseline PRIVATE gme_static)
