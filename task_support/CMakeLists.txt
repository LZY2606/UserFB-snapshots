cmake_minimum_required(VERSION 3.15)
project(nes_dac_task LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(GME_ZLIB OFF CACHE BOOL "" FORCE)
set(GME_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(GME_BUILD_STATIC ON CACHE BOOL "" FORCE)
set(GME_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GME_BUILD_TESTING OFF CACHE BOOL "" FORCE)
foreach(format AY GBS GYM HES KSS SAP SPC VGM)
  set(USE_GME_${format} OFF CACHE BOOL "" FORCE)
endforeach()
set(USE_GME_NSF ON CACHE BOOL "" FORCE)
set(USE_GME_NSFE ON CACHE BOOL "" FORCE)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/.." gme-upstream)
add_executable(task_baseline baseline.cpp)
target_link_libraries(task_baseline PRIVATE gme_static)
foreach(kind basics basics_mem)
  add_executable(old_${kind} "../test/${kind}.c" ../test/Wave_Writer.cpp)
  target_link_libraries(old_${kind} PRIVATE gme_static)
endforeach()
add_executable(compare_wav compare_wav.cpp)
if(TASK_TARGET)
  add_executable(task_target driver.cpp ../test/nes_nonlinear_target.cpp)
  target_link_libraries(task_target PRIVATE gme_static)
  target_include_directories(task_target PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/..")
endif()
