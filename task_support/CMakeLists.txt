cmake_minimum_required(VERSION 3.15)
project(task_dumb C)
set(CMAKE_C_STANDARD 99)
get_filename_component(DUMB_ROOT "${CMAKE_CURRENT_LIST_DIR}/.." ABSOLUTE)
file(GLOB core CONFIGURE_DEPENDS "${DUMB_ROOT}/src/core/*.c")
file(GLOB helpers CONFIGURE_DEPENDS "${DUMB_ROOT}/src/helpers/*.c")
file(GLOB it CONFIGURE_DEPENDS "${DUMB_ROOT}/src/it/*.c")
add_library(dumb STATIC ${core} ${helpers} ${it})
target_include_directories(dumb PUBLIC "${DUMB_ROOT}/include")
target_compile_definitions(dumb PRIVATE _FILE_OFFSET_BITS=64)
