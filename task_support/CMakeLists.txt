cmake_minimum_required(VERSION 3.14)
project(MotionMatchingHeadless LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(root "${CMAKE_CURRENT_SOURCE_DIR}/..")
set(engine "${root}/Horde3D/Source/Horde3DEngine")
set(H3D_USE_NULL 1)
configure_file("${engine}/config.h.in" "${CMAKE_BINARY_DIR}/config.h")
configure_file("${engine}/egExtensions_auto_include.h.in" "${CMAKE_BINARY_DIR}/egExtensions_auto_include.h")
configure_file("${engine}/egExtensions_auto_install.h.in" "${CMAKE_BINARY_DIR}/egExtensions_auto_install.h")
# Build actual engine implementations; Null is the upstream CPU-only device.
file(GLOB sources CONFIGURE_DEPENDS "${engine}/*.cpp")
list(FILTER sources EXCLUDE REGEX "(egRendererBaseGL2|egRendererBaseGL4|egRendererBaseGLES3|utOpenGL|utOpenGLES3)\\.cpp$")
add_library(Horde3D STATIC ${sources})
target_compile_definitions(Horde3D PUBLIC CMAKE H3D_STATIC_LIBS RAPIDXML_NO_EXCEPTIONS)
target_include_directories(Horde3D PUBLIC "${engine}" "${root}/Horde3D/Source/Shared" "${root}/Horde3D/Bindings/C++" "${root}/Horde3D" "${CMAKE_BINARY_DIR}")
add_executable(motion_baseline baseline.cpp)
target_link_libraries(motion_baseline PRIVATE Horde3D)
