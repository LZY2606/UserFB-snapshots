cmake_minimum_required(VERSION 3.5)
project(task_checks LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(DISABLE_PKGCONFIG ON CACHE BOOL "" FORCE)
set(DISABLE_CMAKE_CONFIG ON CACHE BOOL "" FORCE)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../deps/libebml" ebml EXCLUDE_FROM_ALL)
file(GLOB production_sources CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/../src/*.cpp")
add_library(task_production STATIC ${production_sources})
include(GenerateExportHeader)
generate_export_header(task_production EXPORT_FILE_NAME "${CMAKE_CURRENT_BINARY_DIR}/matroska_export.h" EXPORT_MACRO_NAME MATROSKA_DLL_API STATIC_DEFINE MATROSKA_STATIC_DEFINE)
target_compile_definitions(task_production PUBLIC MATROSKA_STATIC_DEFINE)
target_include_directories(task_production PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/.." "${CMAKE_CURRENT_BINARY_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/ebml")
target_link_libraries(task_production PUBLIC ebml)
add_executable(baseline baseline.cpp)
target_link_libraries(baseline PRIVATE task_production)
