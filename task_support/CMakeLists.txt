cmake_minimum_required(VERSION 3.15)
project(WeightedCorridorHarness LANGUAGES CXX)
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/.." ABSOLUTE)
file(GLOB DETOUR_SOURCES CONFIGURE_DEPENDS "${ROOT}/Detour/Source/*.cpp")
file(GLOB CROWD_SOURCES CONFIGURE_DEPENDS "${ROOT}/DetourCrowd/Source/*.cpp")
add_library(task_detour STATIC ${DETOUR_SOURCES} ${CROWD_SOURCES})
set_target_properties(task_detour PROPERTIES CXX_STANDARD 98 CXX_STANDARD_REQUIRED YES)
target_compile_options(task_detour PRIVATE -fno-exceptions -fno-rtti)
target_include_directories(task_detour PUBLIC "${ROOT}/Detour/Include" "${ROOT}/DetourCrowd/Include")
add_executable(starting_point baseline.cpp)
target_link_libraries(starting_point PRIVATE task_detour)
add_library(task_catch STATIC "${ROOT}/Tests/Contrib/catch2/catch_amalgamated.cpp")
set_target_properties(task_catch PROPERTIES CXX_STANDARD 20 CXX_STANDARD_REQUIRED YES)
target_compile_definitions(task_catch PRIVATE CATCH_AMALGAMATED_CUSTOM_MAIN)
target_include_directories(task_catch PUBLIC "${ROOT}/Tests/Contrib")
add_executable(old_tests catch_main.cpp "${ROOT}/Tests/Detour/Tests_Detour.cpp" "${ROOT}/Tests/DetourCrowd/Tests_DetourPathCorridor.cpp")
set_target_properties(old_tests PROPERTIES CXX_STANDARD 20 CXX_STANDARD_REQUIRED YES)
target_link_libraries(old_tests PRIVATE task_detour task_catch)
if(TASK_TARGET)
    # This exact new file is mandatory. No optional branch or alternative path.
    add_executable(weighted_tests catch_main.cpp "${ROOT}/Tests/Detour/Tests_WeightedCorridor.cpp")
    set_target_properties(weighted_tests PROPERTIES CXX_STANDARD 20 CXX_STANDARD_REQUIRED YES)
    target_link_libraries(weighted_tests PRIVATE task_detour task_catch)
endif()
