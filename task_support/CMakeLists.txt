cmake_minimum_required(VERSION 3.22.1)
project(ConstantArgumentTask LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(BUILD_EXTERNAL OFF CACHE BOOL "" FORCE)
set(ENABLE_OPT OFF CACHE BOOL "" FORCE)
set(ENABLE_HLSL OFF CACHE BOOL "" FORCE)
set(GLSLANG_TESTS OFF CACHE BOOL "" FORCE)
set(GLSLANG_ENABLE_INSTALL OFF CACHE BOOL "" FORCE)
set(ENABLE_GLSLANG_BINARIES ON CACHE BOOL "" FORCE)
set(ENABLE_PCH OFF CACHE BOOL "" FORCE)
set(ENABLE_SPIRV ON CACHE BOOL "" FORCE)
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/.." upstream)
add_executable(task_baseline baseline.cpp)
target_include_directories(task_baseline PRIVATE "${CMAKE_CURRENT_LIST_DIR}/..")
target_link_libraries(task_baseline PRIVATE glslang glslang-default-resource-limits)
set_target_properties(task_baseline PROPERTIES OUTPUT_NAME baseline)
if(EXISTS "${CMAKE_CURRENT_LIST_DIR}/../Test/constant_argument_specialization.cpp")
  set_source_files_properties(driver.c PROPERTIES LANGUAGE CXX)
  add_executable(task_target driver.c ../Test/constant_argument_specialization.cpp)
  target_include_directories(task_target PRIVATE "${CMAKE_CURRENT_LIST_DIR}/..")
  target_link_libraries(task_target PRIVATE glslang glslang-default-resource-limits)
  set_target_properties(task_target PROPERTIES OUTPUT_NAME target)
endif()
