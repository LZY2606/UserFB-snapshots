cmake_minimum_required(VERSION 3.16)
project(otio_editorial LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(PYTHON OFF CACHE BOOL "" FORCE)
set(PYBIND11 OFF CACHE BOOL "" FORCE)
set(IMATH_INSTALL OFF CACHE BOOL "" FORCE)
set(IMATH_INSTALL_PKG_CONFIG OFF CACHE BOOL "" FORCE)
add_subdirectory(vendor/Imath)
file(GLOB OTIO_SOURCES CONFIGURE_DEPENDS src/opentimelineio/*.cpp src/opentimelineio/algo/*.cpp)
add_library(opentime STATIC src/opentime/errorStatus.cpp src/opentime/rationalTime.cpp)
target_include_directories(opentime PUBLIC src)
target_compile_definitions(opentime PUBLIC OPENTIME_STATIC)
add_library(opentimelineio STATIC ${OTIO_SOURCES})
target_compile_definitions(opentimelineio PUBLIC OTIO_STATIC)
target_include_directories(opentimelineio PUBLIC src vendor/Imath/src PRIVATE vendor/rapidjson/include)
target_link_libraries(opentimelineio PUBLIC opentime Imath::Imath)
add_executable(baseline task_support/baseline.cpp)
target_link_libraries(baseline PRIVATE opentimelineio)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/task_support/slice_test.cpp")
 set_source_files_properties(task_support/driver.c PROPERTIES LANGUAGE CXX)
 add_executable(target task_support/driver.c task_support/slice_test.cpp)
 target_include_directories(target PRIVATE .)
 target_link_libraries(target PRIVATE opentimelineio)
endif()
