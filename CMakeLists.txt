cmake_minimum_required(VERSION 3.16)
project(libmad_task C)
set(CMAKE_C_STANDARD 99)
include(TestBigEndian)
test_big_endian(IS_BIG_ENDIAN)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/config.h" "#define HAVE_ASSERT_H 1\n#define HAVE_LIMITS_H 1\n#define HAVE_UNISTD_H 1\n")
if(IS_BIG_ENDIAN)
 file(APPEND "${CMAKE_CURRENT_BINARY_DIR}/config.h" "#define WORDS_BIGENDIAN 1\n")
endif()
configure_file(task_support/public_mad.h "${CMAKE_CURRENT_BINARY_DIR}/mad.h" COPYONLY)
file(GLOB PROD CONFIGURE_DEPENDS RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/*.c")
list(REMOVE_ITEM PROD minimad.c)
add_library(mad_task STATIC ${PROD})
target_include_directories(mad_task PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_definitions(mad_task PUBLIC HAVE_CONFIG_H FPM_64BIT OPT_ACCURACY)
add_executable(baseline task_support/baseline.c)
target_link_libraries(baseline PRIVATE mad_task m)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/examples/mad_adu.c")
 add_executable(mad_adu_example examples/mad_adu.c)
 target_link_libraries(mad_adu_example PRIVATE mad_task m)
endif()
